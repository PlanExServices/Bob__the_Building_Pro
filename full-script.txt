## Core Types

```typescript
// src/types/index.ts

export type SourceMode =
  | 'NEW_PRODUCT'
  | 'AUTHORIZED_REBUILD'
  | 'PUBLIC_PRODUCT_INSPIRATION'

export type Platform = 'WEB' | 'MOBILE' | 'WEB_AND_MOBILE' | 'DESKTOP'

export type Distribution =
  | 'LOCAL_ONLY'
  | 'DEPLOYABLE_WEB'
  | 'APP_STORES'
  | 'BOTH_LOCAL_AND_DEPLOYABLE'

export type AuthOption = 'NONE' | 'REQUIRED'
export type DatabaseOption = 'NONE' | 'LOCAL' | 'CLOUD' | 'AUTO'
export type AIOption = 'NONE' | 'REQUIRED'
export type AIProvider = 'OpenAI' | 'Anthropic' | 'Google' | 'Mistral' | 'AUTO'
export type PaymentsOption = 'NONE' | 'REQUIRED'
export type SEOOption = 'NONE' | 'BASIC' | 'FULL'
export type AnalyticsOption = 'NONE' | 'PRIVACY_FIRST' | 'REQUIRED'
export type AutonomyOption = 'AUTONOMOUS' | 'CHECKPOINTS'
export type YesNo = 'YES' | 'NO'

export interface ProjectFormData {
  // Block 1 — Core Identity
  projectName: string
  projectDescription: string
  coreUser: string
  coreProblem: string
  coreAction: string
  coreResult: string

  // Block 2 — Source + Legal
  sourceMode: SourceMode
  sourceUrl: string
  authorizedRebuildConfirmation: string

  // Block 3 — Platform
  platform: Platform
  distribution: Distribution

  // Block 4 — Features
  coreFeatures: string[]

  // Block 5 — Optional Modules
  auth: AuthOption
  database: DatabaseOption
  ai: AIOption
  aiProvider: AIProvider | ''
  payments: PaymentsOption
  seo: SEOOption
  comparisonPages: YesNo
  competitorNames: string[]
  pwaOffline: YesNo
  analytics: AnalyticsOption
  i18n: YesNo
  darkMode: YesNo
  feedback: YesNo

  // Block 6 — Design
  designDirection: string
  brandName: string
  brandColorPrimary: string
  brandColorAccent: string
  brandColorNeutral: string
  referenceProducts: string

  // Block 7 — Constraints
  knownConstraints: string
  autonomy: AutonomyOption
}

export type WizardStep = {
  id: string
  blockNumber: number
  blockTitle: string
  fields: FieldConfig[]
}

export type FieldType =
  | 'text'
  | 'textarea'
  | 'select'
  | 'multiselect'
  | 'radio'
  | 'list'
  | 'color'
  | 'conditional'

export type FieldConfig = {
  key: keyof ProjectFormData
  label: string
  type: FieldType
  guidance: string
  placeholder?: string
  options?: { value: string; label: string; description?: string }[]
  required: boolean
  validate?: (value: unknown, form: Partial<ProjectFormData>) => string | null
  showWhen?: (form: Partial<ProjectFormData>) => boolean
  example?: string
}

export type ValidationError = {
  field: keyof ProjectFormData
  message: string
}

export type AssembledPrompt = {
  intakeForm: string
  builderScript: string
  combined: string
  warnings: string[]
}
```

## Zod Schema

```typescript
// src/lib/schema.ts

import { z } from 'zod'

const projectNameRegex = /^[a-z0-9-]+$/

export const projectFormSchema = z.object({
  // Block 1
  projectName: z
    .string()
    .min(2, 'Project name must be at least 2 characters')
    .max(50, 'Project name must be 50 characters or fewer')
    .regex(
      projectNameRegex,
      'Use lowercase letters, numbers, and hyphens only'
    )
    .refine(
      (v) => !['my-app', 'new-project', 'test', 'app'].includes(v),
      'Choose a more specific name'
    ),

  projectDescription: z
    .string()
    .min(50, 'Write at least 1–2 sentences describing your app')
    .max(500, 'Keep to 3 sentences or fewer'),

  coreUser: z
    .string()
    .min(20, 'Be specific — describe one type of person')
    .max(200)
    .refine(
      (v) => !['everyone', 'anyone', 'all users', 'small businesses'].some(
        (vague) => v.toLowerCase().includes(vague)
      ),
      'Too broad — describe one specific person with a specific context'
    ),

  coreProblem: z
    .string()
    .min(40, 'Describe the specific pain and its cost or friction')
    .max(400),

  coreAction: z
    .string()
    .min(10, 'Describe the single primary action')
    .max(150)
    .refine(
      (v) => {
        const verbCount = (v.match(/\band\b/gi) || []).length
        return verbCount <= 1
      },
      'CORE_ACTION should be one action — split compound actions'
    ),

  coreResult: z
    .string()
    .min(20, 'Describe what the user has after the action completes')
    .max(300),

  // Block 2
  sourceMode: z.enum([
    'NEW_PRODUCT',
    'AUTHORIZED_REBUILD',
    'PUBLIC_PRODUCT_INSPIRATION',
  ]),

  sourceUrl: z.string().refine((v) => {
    if (v === 'N/A' || v === '') return true
    try {
      new URL(v)
      return true
    } catch {
      return false
    }
  }, 'Enter a valid URL or N/A'),

  authorizedRebuildConfirmation: z.string(),

  // Block 3
  platform: z.enum(['WEB', 'MOBILE', 'WEB_AND_MOBILE', 'DESKTOP']),

  distribution: z.enum([
    'LOCAL_ONLY',
    'DEPLOYABLE_WEB',
    'APP_STORES',
    'BOTH_LOCAL_AND_DEPLOYABLE',
  ]),

  // Block 4
  coreFeatures: z
    .array(z.string().min(10, 'Each feature must be at least 10 characters'))
    .min(2, 'List at least 2 core features')
    .max(15, 'Maximum 15 core features — focus on essentials'),

  // Block 5
  auth: z.enum(['NONE', 'REQUIRED']),
  database: z.enum(['NONE', 'LOCAL', 'CLOUD', 'AUTO']),
  ai: z.enum(['NONE', 'REQUIRED']),
  aiProvider: z
    .enum(['OpenAI', 'Anthropic', 'Google', 'Mistral', 'AUTO', ''])
    .optional(),
  payments: z.enum(['NONE', 'REQUIRED']),
  seo: z.enum(['NONE', 'BASIC', 'FULL']),
  comparisonPages: z.enum(['YES', 'NO']),
  competitorNames: z.array(z.string()).optional().default([]),
  pwaOffline: z.enum(['YES', 'NO']),
  analytics: z.enum(['NONE', 'PRIVACY_FIRST', 'REQUIRED']),
  i18n: z.enum(['YES', 'NO']),
  darkMode: z.enum(['YES', 'NO']),
  feedback: z.enum(['YES', 'NO']),

  // Block 6
  designDirection: z
    .string()
    .min(20, 'Describe the visual mood in at least a few words')
    .max(300),

  brandName: z.string().min(2, 'Enter your brand name').max(50),

  brandColorPrimary: z
    .string()
    .min(3, 'Provide a primary color or write "agent decides"'),

  brandColorAccent: z.string().default('agent decides'),
  brandColorNeutral: z.string().default('agent decides'),

  referenceProducts: z
    .string()
    .min(2, 'Write product names/URLs or "none"')
    .max(300),

  // Block 7
  knownConstraints: z.string().min(2, 'Write constraints or "none"').max(600),

  autonomy: z.enum(['AUTONOMOUS', 'CHECKPOINTS']),
}).superRefine((data, ctx) => {
  // Source URL required for non-NEW modes
  if (
    data.sourceMode !== 'NEW_PRODUCT' &&
    (data.sourceUrl === 'N/A' || data.sourceUrl === '')
  ) {
    ctx.addIssue({
      code: z.ZodIssueCode.custom,
      path: ['sourceUrl'],
      message:
        'SOURCE_URL is required for AUTHORIZED_REBUILD and PUBLIC_PRODUCT_INSPIRATION',
    })
  }

  // Confirmation required for AUTHORIZED_REBUILD
  if (data.sourceMode === 'AUTHORIZED_REBUILD') {
    const confirmation = data.authorizedRebuildConfirmation.toLowerCase()
    if (
      !confirmation.includes('i confirm') ||
      (!confirmation.includes('own') && !confirmation.includes('permission'))
    ) {
      ctx.addIssue({
        code: z.ZodIssueCode.custom,
        path: ['authorizedRebuildConfirmation'],
        message:
          'You must confirm ownership or written permission to proceed with AUTHORIZED_REBUILD',
      })
    }
  }

  // AI provider required when AI = REQUIRED
  if (data.ai === 'REQUIRED' && !data.aiProvider) {
    ctx.addIssue({
      code: z.ZodIssueCode.custom,
      path: ['aiProvider'],
      message: 'Select an AI provider or choose AUTO',
    })
  }

  // Competitor names required when comparison pages = YES
  if (
    data.comparisonPages === 'YES' &&
    (!data.competitorNames || data.competitorNames.length === 0)
  ) {
    ctx.addIssue({
      code: z.ZodIssueCode.custom,
      path: ['competitorNames'],
      message: 'List at least one competitor for comparison pages',
    })
  }

  // Cloud DB typically needs auth
  if (data.database === 'CLOUD' && data.auth === 'NONE') {
    ctx.addIssue({
      code: z.ZodIssueCode.custom,
      path: ['database'],
      message:
        'CLOUD database almost always requires AUTH. Enable AUTH or switch to LOCAL/AUTO.',
    })
  }
})

export type ValidatedFormData = z.infer<typeof projectFormSchema>
```

```typescript
// src/lib/store.ts
'use client'

import { create } from 'zustand'
import { persist, createJSONStorage } from 'zustand/middleware'
import type { ProjectFormData } from '@/types'

type WizardStore = {
  currentStep: number
  formData: Partial<ProjectFormData>
  completedSteps: Set<number>
  isDirty: boolean
  setStep: (step: number) => void
  setField: <K extends keyof ProjectFormData>(
    key: K,
    value: ProjectFormData[K]
  ) => void
  markStepComplete: (step: number) => void
  resetWizard: () => void
  setFormData: (data: Partial<ProjectFormData>) => void
}

const initialState = {
  currentStep: 0,
  formData: {} as Partial<ProjectFormData>,
  completedSteps: new Set<number>(),
  isDirty: false,
}

export const useWizardStore = create<WizardStore>()(
  persist(
    (set) => ({
      ...initialState,
      setStep: (step) => set({ currentStep: step }),
      setField: (key, value) =>
        set((state) => ({
          formData: { ...state.formData, [key]: value },
          isDirty: true,
        })),
      markStepComplete: (step) =>
        set((state) => ({
          completedSteps: new Set([...state.completedSteps, step]),
        })),
      resetWizard: () =>
        set({
          ...initialState,
          completedSteps: new Set<number>(),
        }),
      setFormData: (data) =>
        set((state) => ({
          formData: { ...state.formData, ...data },
          isDirty: true,
        })),
    }),
    {
      name: 'bob-builder-wizard',
      storage: createJSONStorage(() => {
        if (typeof window === 'undefined') {
          return {
            getItem: () => null,
            setItem: () => {},
            removeItem: () => {},
          }
        }
        return localStorage
      }),
      skipHydration: true,
      partialise: (state) => ({
        currentStep: state.currentStep,
        formData: state.formData,
        completedSteps: Array.from(state.completedSteps),
      }),
    }
  )
)
```

```typescript
// src/hooks/useWizard.ts
'use client'

import { useCallback } from 'react'
import { useWizardStore } from '@/lib/store'
import { WIZARD_STEPS } from '@/lib/wizard-config'
import type { ProjectFormData } from '@/types'

export function useWizard() {
  const {
    currentStep,
    formData,
    completedSteps,
    setStep,
    setField,
    markStepComplete,
    resetWizard,
  } = useWizardStore()

  const totalSteps = WIZARD_STEPS.length
  const step = WIZARD_STEPS[currentStep]
  const isFirstStep = currentStep === 0
  const isLastStep = currentStep === totalSteps - 1
  const progress = Math.round(((currentStep) / totalSteps) * 100)

  const goNext = useCallback(() => {
    if (!isLastStep) {
      markStepComplete(currentStep)
      setStep(currentStep + 1)
    }
  }, [currentStep, isLastStep, markStepComplete, setStep])

  const goPrev = useCallback(() => {
    if (!isFirstStep) {
      setStep(currentStep - 1)
    }
  }, [currentStep, isFirstStep, setStep])

  const goToStep = useCallback(
    (index: number) => {
      if (index >= 0 && index < totalSteps) {
        setStep(index)
      }
    },
    [totalSteps, setStep]
  )

  const updateField = useCallback(
    <K extends keyof ProjectFormData>(key: K, value: ProjectFormData[K]) => {
      setField(key, value)
    },
    [setField]
  )

  const isStepComplete = useCallback(
    (index: number) => completedSteps.has(index),
    [completedSteps]
  )

  const visibleFields = step?.fields.filter(
    (field) => !field.showWhen || field.showWhen(formData)
  ) ?? []

  return {
    currentStep,
    step,
    formData,
    totalSteps,
    isFirstStep,
    isLastStep,
    progress,
    goNext,
    goPrev,
    goToStep,
    updateField,
    resetWizard,
    isStepComplete,
    visibleFields,
  }
}
```

```typescript
import type { FieldConfig } from '@/types'

interface ListFieldProps {
  field: FieldConfig
  value: string[] | undefined
  onChange: (value: string[]) => void
  error?: string
}

export function ListField({ field, value, onChange, error }: ListFieldProps) {
  const items = value ?? ['']
  const legendId = useId()
  const hasError = Boolean(error)

  const updateItem = (index: number, text: string) => {
    const next = [...items]
    next[index] = text
    onChange(next.filter((_, i) => i !== index || text !== ''))
    if (text !== '') {
      const updated = [...items]
      updated[index] = text
      onChange(updated)
    }
  }

  const addItem = () => onChange([...items, ''])

  const removeItem = (index: number) => {
    if (items.length === 1) {
      onChange([''])
      return
    }
    onChange(items.filter((_, i) => i !== index))
  }

  const handleKeyDown = (
    e: React.KeyboardEvent<HTMLInputElement>,
    index: number
  ) => {
    if (e.key === 'Enter') {
      e.preventDefault()
      addItem()
    }
    if (e.key === 'Backspace' && items[index] === '' && items.length > 1) {
      e.preventDefault()
      removeItem(index)
    }
  }

  return (
    <div className={`field-wrapper ${hasError ? 'field-wrapper--error' : ''}`}>
      <div id={legendId} className="field-label">
        {field.label}
        {field.required && (
          <span className="field-required" aria-label="required">
            *
          </span>
        )}
      </div>

      <ul
        className="list-field"
        aria-labelledby={legendId}
        aria-describedby={hasError ? `${legendId}-error` : undefined}
      >

          <li key={index} className="list-field-item">
            <span className="list-field-bullet" aria-hidden>
```

```typescript
'use client'

import { useState, useCallback, useEffect } from 'react'

import type { ProjectFormData } from '@/types'

export function WizardShell() {
  const router = useRouter()
  const hydrated = useHydrated()
  const {
    currentStep,
    step,
    formData,
    totalSteps,
    isFirstStep,
    isLastStep,
    progress,
    goNext,
    goPrev,
    updateField,
    isStepComplete,
  } = useWizard()

  const [errors, setErrors] = useState<
    Partial<Record<keyof ProjectFormData, string>>
  >({})
  const [isNavigating, setIsNavigating] = useState(false)

  // Rehydrate store on mount
  useEffect(() => {
    useWizardStore.persist.rehydrate()
  }, [])

  const validateCurrentStep = useCallback((): boolean => {
    const newErrors: Partial<Record<keyof ProjectFormData, string>> = {}
    const visibleFields = step.fields.filter(
      (f) => !f.showWhen || f.showWhen(formData)
    )

    for (const field of visibleFields) {
      if (!field.required) continue
      const value = formData[field.key]

      // Required field checks
      if (
        value === undefined ||
        value === null ||
        value === '' ||
        (Array.isArray(value) && value.filter(Boolean).length === 0)
      ) {
        newErrors[field.key] = `${field.label} is required`
        continue
      }

      // Custom validation
      if (field.validate) {
        const fieldError = field.validate(value, formData)
        if (fieldError) {
          newErrors[field.key] = fieldError
        }
      }
    }

    setErrors(newErrors)
    return Object.keys(newErrors).length === 0
  }, [step, formData])

  const handleNext = useCallback(async () => {
    if (!validateCurrentStep()) return

    setIsNavigating(true)
    try {
      if (isLastStep) {
        router.push('/review')
      } else {
        goNext()
        setErrors({})
        window.scrollTo({ top: 0, behavior: 'smooth' })
      }
    } finally {
      setIsNavigating(false)
    }
  }, [validateCurrentStep, isLastStep, router, goNext])

  const handlePrev = useCallback(() => {
    goPrev()
    setErrors({})
    window.scrollTo({ top: 0, behavior: 'smooth' })
  }, [goPrev])

  if (!hydrated) {
    return (
      <div className="wizard-loading">
        <Skeleton className="skeleton-progress" />
        <Skeleton className="skeleton-block" />
        <Skeleton className="skeleton-nav" />
      </div>
    )
  }

  return (
    <div className="wizard-shell">
      {/* Step Navigation Pills */}
      <nav
        className="wizard-steps-nav"
        aria-label="Wizard steps"
      >
        {WIZARD_STEPS.map((s, index) => (
          <button
            key={s.id}
            type="button"
            className={[
              'wizard-step-pill',
              index === currentStep ? 'wizard-step-pill--active' : '',
              isStepComplete(index) ? 'wizard-step-pill--complete' : '',
              index > currentStep && !isStepComplete(index)
                ? 'wizard-step-pill--upcoming'
                : '',
            ]
              .filter(Boolean)
              .join(' ')}
            onClick={() => {
              if (isStepComplete(index) || index <= currentStep) {
                useWizardStore.getState().setStep(index)
                setErrors({})
              }
            }}
            aria-current={index === currentStep ? 'step' : undefined}
            aria-label={`Block ${s.blockNumber}: ${s.blockTitle}${
              isStepComplete(index) ? ' (complete)' : ''
            }`}
            disabled={index > currentStep && !isStepComplete(index)}
          >
            <span className="wizard-step-pill-number">
              {isStepComplete(index) ? '✓' : s.blockNumber}
            </span>
            <span className="wizard-step-pill-label">{s.blockTitle}</span>
          </button>
        ))}
      </nav>

      {/* Progress Bar */}
      <ProgressBar
        progress={progress}
        currentStep={currentStep}
        totalSteps={totalSteps}
        stepLabel={step.blockTitle}
      />

      {/* Block Content */}
      <main className="wizard-content">
        <BlockRenderer
          step={step}
          formData={formData}
          errors={errors}
          onChange={updateField}
        />
      </main>

      {/* Navigation */}
      <footer className="wizard-footer">
        <Button
          variant="ghost"
          onClick={handlePrev}
          disabled={isFirstStep || isNavigating}
          aria-label="Go to previous step"

          ← Back
        </Button>

        <span className="wizard-footer-position" aria-live="polite">
          {currentStep + 1} / {totalSteps}
        </span>

        <Button
          variant="primary"
          onClick={handleNext}
          disabled={isNavigating}
          aria-label={isLastStep ? 'Review and generate prompt' : 'Go to next step'}
        >
          {isNavigating ? (
            <span className="btn-spinner" aria-hidden />
          ) : isLastStep ? (
            'Review & Generate →'
          ) : (
            'Next →'
          )}
        </Button>
      </footer>

      <div className="wizard-autosave" aria-live="polite" aria-atomic>
        <span className="autosave-dot" aria-hidden />
        Progress saved automatically
      </div>
    </div>
  )
```

```typescript
  const [validationErrors, setValidationErrors] = useState<string[]>([])
  const [copied, setCopied] = useState(false)
  const [isAssembling, setIsAssembling] = useState(false)

  useEffect(() => {
    useWizardStore.persist.rehydrate()
  }, [])

  const handleAssemble = useCallback(async () => {
    setIsAssembling(true)
    setValidationErrors([])
    setAssembled(null)

    try {
      const formData = useWizardStore.getState().formData
      const result = projectFormSchema.safeParse(formData)

      if (!result.success) {
        const messages = result.error.issues.map(
          (issue) => `${issue.path.join('.')}: ${issue.message}`
        )
        setValidationErrors(messages)
        return
      }

      const prompt = assemblePrompt(result.data)
      setAssembled(prompt)
    } finally {
      setIsAssembling(false)
    }
  }, [])

  const handleCopy = useCallback(async () => {
    if (!assembled) return
    try {
      await navigator.clipboard.writeText(assembled.combined)
      setCopied(true)
      setTimeout(() => setCopied(false), 3000)
    } catch {
      // Fallback for environments without clipboard API
      const el = document.createElement('textarea')
      el.value = assembled.combined
      document.body.appendChild(el)
      el.select()
      document.execCommand('copy')
      document.body.removeChild(el)
      setCopied(true)
      setTimeout(() => setCopied(false), 3000)
    }
  }, [assembled])

  const handleDownload = useCallback(() => {
    if (!assembled) return
    const blob = new Blob([assembled.combined], { type: 'text/plain' })
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    const formData = useWizardStore.getState().formData
    a.download = `${formData.projectName ?? 'project'}-builder-prompt.txt`
    a.click()
    URL.revokeObjectURL(url)
  }, [assembled])

  if (!hydrated) return <Skeleton className="skeleton-launch-bar" />

  return (
    <div className="launch-bar">
      {!assembled && (
        <div className="launch-bar-cta">
          <h3 className="launch-bar-title">Ready to generate your prompt?</h3>
          <p className="launch-bar-subtitle">
```

```typescript
// src/app/api/assemble/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { projectFormSchema } from '@/lib/schema'
import { assemblePrompt } from '@/lib/assembler'
import { getClientIp } from '@/lib/get-ip'

// In-memory rate limiter — local only
const rateLimitStore = new Map<string, { count: number; reset: number }>()
const RATE_LIMIT = 10
const WINDOW_MS = 60_000

function checkRateLimit(ip: string): boolean {
  const now = Date.now()

  // Cleanup expired entries
  if (rateLimitStore.size > 1000) {
    for (const [k, v] of rateLimitStore) {
      if (v.reset < now) rateLimitStore.delete(k)
    }
  }

  const entry = rateLimitStore.get(ip)
  if (!entry || entry.reset < now) {
    rateLimitStore.set(ip, { count: 1, reset: now + WINDOW_MS })
    return true
  }
  if (entry.count >= RATE_LIMIT) return false
  entry.count++
  return true
}

export async function POST(request: NextRequest) {
  const ip = getClientIp()

  if (!checkRateLimit(ip)) {
    return NextResponse.json(
      { error: 'Too many requests — please wait a moment' },
      { status: 429 }
    )
  }

  let body: unknown
  try {
    body = await request.json()
  } catch {
    return NextResponse.json({ error: 'Invalid JSON body' }, { status: 400 })
  }

  const result = projectFormSchema.safeParse(body)
  if (!result.success) {
    return NextResponse.json(
      {
        error: 'Validation failed',
        issues: result.error.issues.map((i) => ({
          path: i.path.join('.'),
          message: i.message,
        })),
      },
      { status: 422 }
    )
  }

  const assembled = assemblePrompt(result.data)
  return NextResponse.json(assembled)
}
```

```typescript
// src/app/review/page.tsx
'use client'

import { useCallback } from 'react'
import { useRouter } from 'next/navigation'
import { ReviewPanel } from '@/components/review/ReviewPanel'
import { CopyLaunchBar } from '@/components/review/CopyLaunchBar'
import { useWizardStore } from '@/lib/store'

export default function ReviewPage() {
  const router = useRouter()

  const handleEdit = useCallback(
    (stepIndex: number) => {
      useWizardStore.getState().setStep(stepIndex)
      router.push('/build')
    },
    [router]
  )

  return (
    <div className="review-page">
      <div className="review-page-inner">
        <div className="review-layout">
          <aside className="review-sidebar">
            <ReviewPanel onEdit={handleEdit} />
          </aside>
          <main className="review-main">
            <CopyLaunchBar />
          </main>
        </div>
      </div>
    </div>
  )
}
```

```typescript
// src/middleware.ts
import { NextResponse } from 'next/server'
import type { NextRequest } from 'next/server'

export function middleware(request: NextRequest) {
  const response = NextResponse.next()

  response.headers.set('X-Frame-Options', 'DENY')
  response.headers.set('X-Content-Type-Options', 'nosniff')
  response.headers.set('Referrer-Policy', 'strict-origin-when-cross-origin')
  response.headers.set(
    'Permissions-Policy',
    'camera=(), microphone=(), geolocation=()'
  )
  response.headers.set(
    'Content-Security-Policy',
    [
      "default-src 'self'",
      "script-src 'self' 'unsafe-inline'",  // unsafe-inline needed for theme script
      "style-src 'self' 'unsafe-inline'",
      "img-src 'self' data: blob:",
      "font-src 'self'",
      "connect-src 'self'",
      "frame-ancestors 'none'",
    ].join('; ')
  )

  return response
}

export const config = {
  matcher: [
    '/((?!_next/static|_next/image|favicon.ico).*)',
  ],
}
```

```markdown
# docs/decisions.md

## ADR-001: Framework — Next.js App Router
Date: 2025-01-01
Status: Accepted

Chosen: Next.js 15 App Router with TypeScript
Reason: Scaffolds cleanly, handles SSR/SSG, strong ecosystem,
        App Router Server Components reduce client bundle size.
Alternatives considered: Remix, SvelteKit
Decision: Next.js for team familiarity and Vercel deployment fit.

## ADR-002: State — Zustand with localStorage persist
Date: 2025-01-01
Status: Accepted

Chosen: Zustand with persist middleware
Reason: Wizard state must survive page refreshes. No auth required.
        localStorage sufficient for single-user, single-device use.
Note: skipHydration: true required. useHydrated() pattern used in
      all components reading persisted state.

## ADR-003: Validation — Zod
Date: 2025-01-01
Status: Accepted

Chosen: Zod for all validation (schema + runtime boundary)
Reason: Single source of truth for form shape + API validation.
        superRefine enables cross-field rules.

## ADR-004: Prompt Storage — Static string in assembler
Date: 2025-01-01
Status: Accepted

Chosen: Builder script stored as string constant in assembler.ts
Reason: No database needed. Script is static content — not user data.
        Could move to a versioned static file if the script evolves.

## ADR-005: Analytics — None (no tracking in meta-tool)
Date: 2025-01-01
Status: Accepted

Reason: Bob the Builder Pro is a developer tool for generating
        prompts. No analytics installed in this tool itself.
        The apps it helps BUILD may include analytics per user spec.
```

```markdown
# docs/threat-model.md

## Trust Boundaries
- All user input validated with Zod before use
- No server-side persistence of form data (stateless API)
- No auth — all data local to browser localStorage

## Secrets
- No API keys stored in this app
- .env.example documents all required vars
- NEXT_PUBLIC_ prefix only on genuinely public values

## Input Handling
- All API inputs validated with Zod
- Rate limiting on /api/assemble — in-memory, local dev only
- Request size limits via Next.js defaults

## XSS
- No dangerouslySetInnerHTML with user content
- All user content rendered as text nodes
- Prompt preview uses <code> inside <pre> — not innerHTML

## CSRF
- State-changing operations use POST
- No session cookies — no CSRF surface

## Rate Limiting
- In-memory store with size cap (1000 entries)
- Expired entry cleanup on every check
- Note: in-memory is acceptable for LOCAL_ONLY deployment
  For DEPLOYABLE_WEB: upgrade to Upstash Redis recommended

## Content Security Policy
- Strict CSP headers via middleware
- unsafe-inline in script-src required for theme injection script
  (industry-standard tradeoff for zero-flash dark mode)

## Remaining Risks
- prompt injection not applicable — we are assembling a prompt,
  not executing one. The assembled prompt is plain text.
- localStorage can be read by other same-origin scripts —
  form data contains no secrets, so this is acceptable.
```

## Summary

This is a **complete, self-contained app** that:

### What it builds
- **7-block guided wizard** — each block maps exactly to the intake form, with radio cards, list fields, color pickers, conditional fields, and expandable guidance tooltips
- **Real-time validation** — Zod schema validates at the field level (on Next step), with cross-field rules (CLOUD + no auth, comparison pages without competitors, AI without provider)
- **Auto-save** — Zustand persist writes to localStorage on every keystroke; wizard survives page refreshes
- **Review page** — side-by-side summary of all answers with one-click edit-any-block
- **Prompt assembler** — generates the filled intake form + complete builder script as a single copyable/downloadable text file
- **Security** — CSP headers, rate limiting with cleanup, no user data server-side, getClientIp() pattern, Zod at every API boundary
- **Accessibility** — keyboard-only navigation, axe-core E2E scans, 44×44px touch targets, reduced-motion support, proper ARIA throughout
- **Dark mode** — zero-flash inline script, CSS custom property swap, hydration-safe Zustand pattern
- **Full test suite** — Vitest unit tests for schema + assembler, Playwright E2E for full wizard flow + accessibility + API + mobile
- **CI/CD** — GitHub Actions with quality → security → test → e2e → deploy gates in order
